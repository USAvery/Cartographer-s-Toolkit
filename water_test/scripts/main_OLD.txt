func search_down(obj, filter=null, camel_only=false):
    if not obj:
        return

    var dir_string = utils.dir_string(obj, filter)

    var camel_size = 0

    for line in dir_string.split('\n'):
        if camel_only and line.to_lower() != line:
            camel_size += 1

    if not camel_only or camel_size > 3:
        if dir_string.split('\n').size() > 3:            
            print('\n---' + str(obj) + '---')
            for line in dir_string.split('\n'):
                if camel_only and line.to_lower() == line:
                    continue
                print(line)

    # if 'cam' in str(obj).to_lower():
    #     utils.print_ancestors(obj)
    
    for child in obj.get_children():
        search_down(child, filter, camel_only)
